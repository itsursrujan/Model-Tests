15. SELECT b.branch_city, AVG(a.balance) AS avg_balance
FROM account a
JOIN branch b ON a.branchcode = b.branch_code
GROUP BY b.branch_city;

16. SELECT b.branch_city, COUNT(DISTINCT d.cust_no) AS num_depositors
FROM depositor d
JOIN account a ON d.ac_no = a.account_no
JOIN branch b ON a.branchcode = b.branch_code
GROUP BY b.branch_city;

17. SELECT bc.customer_name
FROM bank_customer bc
JOIN depositor d ON bc.customer_number = d.cust_no
JOIN account a ON d.ac_no = a.account_no
JOIN branch b ON a.branchcode = b.branch_code
WHERE b.branch_city = 'Hubli'
GROUP BY bc.customer_name
HAVING COUNT(a.account_no) <= 1;
